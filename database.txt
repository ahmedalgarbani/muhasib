-- Create views for common queries
CREATE VIEW current_stock AS
SELECT 
    c.id,
    c.name,
    c.barcode_no,
    c.quantity,
    s.name as stock_name,
    c.cost_amount,
    c.sell_amount,
    c.min_stock_level,
    c.max_stock_level
FROM categories c
JOIN stocks s ON c.stock_id = s.id
WHERE c.is_active = 1;

CREATE VIEW customer_balances AS
SELECT 
    cust.id,
    cust.name,
    acc.code as account_code,
    SUM(CASE WHEN jel.credit IS NOT NULL THEN jel.credit ELSE 0 END) as total_credit,
    SUM(CASE WHEN jel.debit IS NOT NULL THEN jel.debit ELSE 0 END) as total_debit,
    (SUM(CASE WHEN jel.debit IS NOT NULL THEN jel.debit ELSE 0 END) - 
     SUM(CASE WHEN jel.credit IS NOT NULL THEN jel.credit ELSE 0 END)) as balance
FROM customers cust
JOIN accounts acc ON cust.account_id = acc.id
LEFT JOIN journal_entry_lines jel ON jel.account_id = acc.id
LEFT JOIN journal_entries je ON je.id = jel.journal_entry_id
WHERE cust.is_active = 1
GROUP BY cust.id, cust.name, acc.code;

CREATE VIEW low_stock_alerts AS
SELECT 
    c.id,
    c.name,
    c.barcode_no,
    c.quantity,
    c.min_stock_level,
    s.name as stock_name
FROM categories c
JOIN stocks s ON c.stock_id = s.id
WHERE c.is_active = 1 
AND c.quantity <= c.min_stock_level
AND c.min_stock_level > 0;
